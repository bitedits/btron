let xhci_ring_enqueue (ring : xhci_ring) (ptr : int) (len : int) (chain : bool) : unit =
  ring.trbs.(ring.enq_idx) <- { data_ptr = ptr; status_len = len; cycle = ring.producer_cycle; chain; kind = TRB_NORMAL };
  let next = ring.enq_idx + 1 in
  if next >= ring.capacity then ( ring.enq_idx <- 0; ring.producer_cycle <- not ring.producer_cycle )
  else ring.enq_idx <- next
