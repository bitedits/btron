(* BTRON wget_tab_rect operational oracle *)
let wget_tab_rect (wnd : wnd) : rect =
  if (wnd.attr land wnd_attr_title) = 0 then rect_empty
  else
    let w = rect_width wnd.bounds in
    let tw = if wnd.tab_width <= 0 || wnd.tab_width > w || (wnd.attr land wnd_attr_compact_tab) = 0
             then w else wnd.tab_width in
    let off_x = if wnd.tab_offset_x < 0 then 0
                else if wnd.tab_offset_x + tw > w then max 0 (w - tw)
                else wnd.tab_offset_x in
    { left = wnd.bounds.left + off_x; top = wnd.bounds.top;
      right = wnd.bounds.left + off_x + tw; bottom = wnd.bounds.top + 28 }

let whit_test_close_btn (wnd : wnd) (x : int) (y : int) : bool =
  let tab_r = wget_tab_rect wnd in
  point_in_rect x y { left = tab_r.right - 24; top = tab_r.top + 6;
                      right = tab_r.right - 6; bottom = tab_r.top + 22 }

Theorem bring_to_top_guarantees_focus_unique : forall target_id ws target,
  find (fun w => Nat.eqb w.(w_id) target_id) ws = Some target ->
  focus_unique (bring_to_top target_id ws).

Theorem bring_to_top_head_id : forall target_id ws target,
  find (fun w => Nat.eqb w.(w_id) target_id) ws = Some target ->
  match bring_to_top target_id ws with
  | [] => False
  | h :: _ => h.(w_id) = target.(w_id) /\ h.(w_focused) = true
  end.
Why we are proving this: HMI determinism: at most one window can be active at any given moment, and the top-most window under the pointer always receives keyboard focus.

Cat. VI, IX Category 4: Event Queue FIFO Invariance & Capacity Bounds

Proves that the TIP input event queue strictly maintains FIFO temporal ordering and never exceeds the bounded capacity of 256 events.

Theorems event_queue_fifo_pair & event_queue_bounded
[x] Proved / Qed 
Theorem event_queue_fifo_pair : forall ev1 ev2,
  match eq_enqueue ev1 [] with
  | Some q1 => match eq_enqueue ev2 q1 with
               | Some q2 => eq_dequeue q2
               | None => None end
  | None => None
  end = Some (ev1, [ev2]).

Theorem event_queue_bounded : forall ev q q',
  eq_enqueue ev q = Some q' ->
  length q' <= EVENT_QUEUE_SIZE.

Theorem event_queue_overflow_rejects : forall ev q,
  length q >= EVENT_QUEUE_SIZE ->
  eq_enqueue ev q = None.
Why we are proving this: Ensures deterministic temporal event delivery from keyboard/mouse interrupts without queue inversion or ring buffer overflows.
